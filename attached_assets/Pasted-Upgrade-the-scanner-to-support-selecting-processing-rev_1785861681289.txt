4. ADD FILES WITHOUT ERASING EXISTING FILES

Use this helper:

function mergeUniqueFiles(
  current: File[],
  incoming: File[],
): File[] {
  const seen = new Set(
    current.map(
      (file) =>
        `${file.name}:${file.size}:${file.lastModified}`,
    ),
  );

  const merged = [...current];

  for (const file of incoming) {
    const key =
      `${file.name}:${file.size}:${file.lastModified}`;

    if (!seen.has(key)) {
      seen.add(key);
      merged.push(file);
    }
  }

  return merged.slice(0, MAX_BATCH_FILES);
}

6. PROCESS WITH A CONCURRENCY LIMIT OF 2

Add this exact helper:

async function runWithConcurrency<T>(
  items: T[],
  limit: number,
  worker: (
    item: T,
    index: number,
  ) => Promise<void>,
): Promise<void> {
  let nextIndex = 0;

  async function runWorker() {
    while (true) {
      const index = nextIndex;
      nextIndex += 1;

      if (index >= items.length) {
        return;
      }

      await worker(items[index], index);
    }
  }

  const workers = Array.from(
    {
      length: Math.min(limit, items.length),
    },
    () => runWorker(),
  );

  await Promise.all(workers);
}

7. PROCESS ONE DOCUMENT INDEPENDENTLY

Keep or create:

const processDoc = async (
  docId: string,
  file: File,
) => {
  setDocs((previous) =>
    previous.map((doc) =>
      doc.id === docId
        ? {
            ...doc,
            status: "processing",
            error: undefined,
            errorStage: undefined,
          }
        : doc,
    ),
  );

  try {
    const token =
      await getToken().catch(() => null);

    const result =
      await scanFile(file, token);

    const {
      resolvedDocType,
      fieldMap,
      institutionName,
      institutionUnknown,
      institutionCategory,
    } = normalizeScanResult(result);

    setDocs((previous) =>
      previous.map((doc) =>
        doc.id === docId
          ? {
              ...doc,
              status: "done",
              result,
              docType: resolvedDocType,
              fields: fieldMap,
              institutionName,
              institutionUnknown,
              institutionCategory,
              accepted: true,
              error: undefined,
              errorStage: undefined,
            }
          : doc,
      ),
    );
  } catch (error) {
    const parsed =
      parseApiError(error);

    setDocs((previous) =>
      previous.map((doc) =>
        doc.id === docId
          ? {
              ...doc,
              status: "error",
              error: parsed.message,
              errorStage: parsed.stage,
              docType: "Unknown",
              fields: {},
              accepted: false,
            }
          : doc,
      ),
    );
  }
};

Do not let one failure throw out of the whole batch.

8. START BATCH PROCESSING

Replace any sequential loop with:

const startProcessing = async () => {
  if (files.length === 0) {
    return;
  }

  setStep("processing");
  setGlobalError("");

  const initialDocs =
    await Promise.all(
      files.map(
        async (file, index) => {
          let preview = "";

          try {
            if (
              file.type.startsWith("image/")
            ) {
              preview =
                URL.createObjectURL(file);
            }
          } catch {
            preview = "";
          }

          let fingerprint:
            | string
            | undefined;

          try {
            fingerprint =
              await fingerprintFile(file);
          } catch {
            fingerprint = undefined;
          }

          return {
            id: crypto.randomUUID(),
            file,
            preview,
            fingerprint,
            status: "pending" as const,
            docType: "Unknown",
            fields: {},
            accepted: true,
            isDuplicate: false,
          };
        },
      ),
    );

  const fingerprintCounts =
    new Map<string, number>();

  for (const doc of initialDocs) {
    if (!doc.fingerprint) {
      continue;
    }

    fingerprintCounts.set(
      doc.fingerprint,
      (fingerprintCounts.get(
        doc.fingerprint,
      ) ?? 0) + 1,
    );
  }

  const existingFingerprints =
    new Set(
      documents
        .map(
          (document) =>
            document.fingerprint,
        )
        .filter(
          (
            fingerprint,
          ): fingerprint is string =>
            Boolean(fingerprint),
        ),
    );

  const preparedDocs =
    initialDocs.map((doc) => ({
      ...doc,
      isDuplicate:
        Boolean(doc.fingerprint) &&
        (
          existingFingerprints.has(
            doc.fingerprint!,
          ) ||
          (
            fingerprintCounts.get(
              doc.fingerprint!,
            ) ?? 0
          ) > 1
        ),
    }));

  setDocs(preparedDocs);

  await runWithConcurrency(
    preparedDocs,
    SCAN_CONCURRENCY,
    async (doc) => {
      if (doc.isDuplicate) {
        setDocs((previous) =>
          previous.map((item) =>
            item.id === doc.id
              ? {
                  ...item,
                  status: "error",
                  accepted: false,
                  errorStage:
                    "duplicate_document",
                  error:
                    "This document appears to be a duplicate.",
                }
              : item,
          ),
        );

        return;
      }

      await processDoc(
        doc.id,
        doc.file,
      );
    },
  );

  setStep("review");
};

const removeDoc = (
  docId: string,
) => {
  setDocs((previous) => {
    const removed =
      previous.find(
        (doc) =>
          doc.id === docId,
      );

    if (removed?.preview) {
      URL.revokeObjectURL(
        removed.preview,
      );

    return previous.filter(
      (doc) =>
        doc.id !== docId,

  });
};
